// Aislamos el scope para este componente específico
document.addEventListener('DOMContentLoaded', () => {
    
    // 1. Identificamos los nodos en el DOM para este componente
    const selectConvivencia = document.getElementById('convivencia_pareja');
    const feedbackConvivencia = document.getElementById('convivencia_pareja_feedback');

    // 2. Estructura de datos (Payload) enfocada en este campo
    const formData = {
        convivencia_pareja: null
    };

    // 3. Validación de seguridad: Comprobamos que el elemento exista antes de inyectar eventos
    if (selectConvivencia) {
        selectConvivencia.addEventListener('change', function(event) {
            // Extracción de datos
            const selectedValue = event.target.value;
            const selectedText = event.target.options[event.target.selectedIndex].text;

            // Actualización del estado (preparando JSON)
            formData.convivencia_pareja = selectedValue;

            // UX: Retroalimentación visual
            if (selectedValue && feedbackConvivencia) {
                // Mensaje en pantalla
                feedbackConvivencia.textContent = `Has seleccionado: ${selectedText}`;
                feedbackConvivencia.classList.add('success');
                
                // Efecto de parpadeo verde en el borde (Microinteracción)
                selectConvivencia.style.borderColor = "#10b981";
                setTimeout(() => {
                    selectConvivencia.style.borderColor = ""; // Retorna al estado por defecto
                }, 1500);
            }

            // Consola para debug / validación de integración backend
            console.log("Dato de Convivencia capturado:", JSON.stringify(formData));
        });
    }
});
