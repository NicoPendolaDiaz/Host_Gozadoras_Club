3. Lógica JavaScript (Modular y Preparada para APIs)
// Usamos el evento DOMContentLoaded para asegurar que el HTML se haya cargado
document.addEventListener('DOMContentLoaded', () => {
    // Referencias a los elementos del DOM
    const selectEstadoCivil = document.getElementById('estado_civil');
    const feedbackText = document.getElementById('estado_civil_feedback');

    // Estructura de datos para simular un envío a Backend/CRM
    const formData = {
        estado_civil: null
    };

    // Escuchador de eventos limpio y modular
    selectEstadoCivil.addEventListener('change', function(event) {
        const selectedValue = event.target.value;
        const selectedText = event.target.options[event.target.selectedIndex].text;

        // 1. Actualizamos nuestro objeto de datos
        formData.estado_civil = selectedValue;

        // 2. Lógica de UI / UX
        if(selectedValue) {
            feedbackText.textContent = `Has seleccionado: ${selectedText}`;
            feedbackText.classList.add('success');
            
            // Aquí quitaríamos clases de error si estuviéramos validando
            selectEstadoCivil.style.borderColor = "#10b981"; // Borde verde temporal
            setTimeout(() => selectEstadoCivil.style.borderColor = "", 1500); // Reseteo
        }

        // 3. (Opcional) Demostración de cómo se estructuraría para enviar por API (JSON)
        console.log("Dato listo para Payload JSON:", JSON.stringify(formData));
        
        // Si tienes una lógica adicional (por ejemplo, si selecciona "Otro" mostrar un input de texto),
        // este es el lugar exacto para detonar esa función:
        /*
        if (selectedValue === 'otro') {
            mostrarInputAdicional();
        }
        */
    });
});
