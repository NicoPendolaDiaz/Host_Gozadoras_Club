document.addEventListener('DOMContentLoaded', () => {
    
    // Referencias del DOM
    const selectPaisResidencia = document.getElementById('pais_residencia');
    const feedbackPaisResidencia = document.getElementById('pais_residencia_feedback');

    // Estructura de datos (Payload)
    const formData = {
        pais_residencia: null
    };

    // Validamos existencia antes de inyectar
    if (selectPaisResidencia) {
        selectPaisResidencia.addEventListener('change', function(event) {
            
            // Extracción
            const selectedValue = event.target.value;
            const selectedText = event.target.options[event.target.selectedIndex].text;

            // Actualización del estado en local
            formData.pais_residencia = selectedValue;

            // Feedback en UI
            if (selectedValue && feedbackPaisResidencia) {
                feedbackPaisResidencia.textContent = `País seleccionado: ${selectedText}`;
                feedbackPaisResidencia.classList.add('success');
                
                // Confirmación visual
                selectPaisResidencia.style.borderColor = "#10b981";
                setTimeout(() => {
                    selectPaisResidencia.style.borderColor = "";
                }, 1500);
            }

            // Consola para debug (Listo para enviar al servidor)
            console.log("Dato 'País' capturado:", JSON.stringify(formData));
        });
    }
});
