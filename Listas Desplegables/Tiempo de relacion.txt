document.addEventListener('DOMContentLoaded', () => {
    
    // 1. Identificamos los elementos específicos en el DOM
    const selectTiempoRelacion = document.getElementById('tiempo_relacion');
    const feedbackTiempoRelacion = document.getElementById('tiempo_relacion_feedback');

    // 2. Payload local (Datos listos para JSON)
    const formData = {
        tiempo_relacion: null
    };

    // 3. Validación y asignación del evento
    if (selectTiempoRelacion) {
        selectTiempoRelacion.addEventListener('change', function(event) {
            
            // Extracción
            const selectedValue = event.target.value;
            const selectedText = event.target.options[event.target.selectedIndex].text;

            // Actualización del estado
            formData.tiempo_relacion = selectedValue;

            // Retroalimentación de la Interfaz (UX)
            if (selectedValue && feedbackTiempoRelacion) {
                feedbackTiempoRelacion.textContent = `Has seleccionado: ${selectedText}`;
                feedbackTiempoRelacion.classList.add('success');
                
                // Microinteracción visual (borde de éxito)
                selectTiempoRelacion.style.borderColor = "#10b981";
                setTimeout(() => {
                    selectTiempoRelacion.style.borderColor = "";
                }, 1500);
            }

            // Consola para debug / validación de integración backend
            console.log("Dato 'Tiempo de relación' capturado:", JSON.stringify(formData));
        });
    }
});
